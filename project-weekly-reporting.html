Business problem

Before this automation, the weekly lead recap reporting process relied heavily on manual steps across multiple tools — extracting data from LeadPortal, preparing Excel reports, and manually drafting partner emails.

The initial preparation stage alone typically took around one hour each week, involving repetitive tasks across different systems that were time-consuming and prone to human error — incorrect file naming, misplaced files, outdated data, or inconsistent report preparation.

  • Filtering partner data and date ranges in LeadPortal manually
  • Manually updating Excel reporting files and recreating pivot tables
  • Manually saving, renaming, and navigating files across folders
  • Manually drafting individual partner emails in Outlook

Solution

I developed a Streamlit-based workflow automation tool that guides the user through the reporting process and automates all repetitive preparation tasks — ensuring the workflow runs in a structured and repeatable way every week.

The application serves as a single-entry point for the entire weekly pipeline, from OAuth authentication through to the Excel report being open and ready for partner distribution.

My contribution

I designed and implemented the full automation layer using Python, Streamlit, Power Query, and VBA — integrating a modern data extraction workflow with the existing Excel-based reporting process without disrupting the team's structure.

  • Implemented Databricks SQL connection using OAuth authentication
  • Automated file archiving and export workflows with dynamic file naming
  • Integrated the workflow with the existing Excel VBA reporting template
  • Added progress indicators, status messages, and dataset previews

SQL_QUERY = """
SELECT DISTINCT
    TRIM(partner_company)                                       AS partner
    , TO_TIMESTAMP(date_created, 'yyyy-MM-dd HH:mm')            AS transaction_date
    , price
    , lead_id
    , company
    , first_name
    , last_name
    , city
    , state
    , zip
    , primary_phone
    , email
    , features
    , how_many_vehicles                                         AS fleet_size
    , vehicle_types
FROM
    schema.bronze.lead_data
WHERE
    partner_company IN (
        '[PARTNER 1]', '[PARTNER 2]', '[PARTNER 3]',
        '[PARTNER 4]', '[PARTNER 5]', '[PARTNER 6]'
    )
    AND TO_DATE(date_created) BETWEEN
        DATE_SUB(NEXT_DAY(CURRENT_DATE(), 'Mon'), 14)
        AND DATE_SUB(NEXT_DAY(CURRENT_DATE(), 'Mon'), 8)
    AND features IS NOT NULL
ORDER BY
    transaction_date DESC
"""

def safe_mkdir(path: str) -> None:
    os.makedirs(path, exist_ok=True)


def get_previous_week_range():
    today = datetime.today()
    this_monday  = today - timedelta(days=today.weekday())
    prev_monday  = this_monday - timedelta(days=7)
    prev_sunday  = this_monday - timedelta(days=1)
    return prev_monday, prev_sunday


def format_month_day_no_zero(dt: datetime, with_year: bool = False) -> str:
    s = dt.strftime("%b %d %Y") if with_year else dt.strftime("%b %d")
    return s.replace(" 0", " ")


def build_output_file_name() -> str:
    prev_monday, prev_sunday = get_previous_week_range()
    start_str = format_month_day_no_zero(prev_monday, with_year=False)
    end_str   = format_month_day_no_zero(prev_sunday, with_year=True)
    return f"Weekly Lead Recap for Week of {start_str} - {end_str}.xlsx"


def run_databricks_query_oauth(server_hostname: str, http_path: str, query: str) -> pd.DataFrame:
    with sql.connect(
        server_hostname=server_hostname,
        http_path=http_path,
        auth_type="oauth",
    ) as connection:
        with connection.cursor() as cursor:
            cursor.execute(query)
            rows = cursor.fetchall()
            cols = [d[0] for d in cursor.description]
    return pd.DataFrame(rows, columns=cols)


def archive_existing_source_files(source_dir: str, archive_dir: str) -> list[dict]:
    source_path  = Path(source_dir)
    archive_path = Path(archive_dir)

    if not source_path.exists():
        raise FileNotFoundError(f"Source folder does not exist: {source_dir}")

    archive_path.mkdir(parents=True, exist_ok=True)
    moved_files = []

    for item in source_path.iterdir():
        if not item.is_file():
            continue
        target_path = archive_path / item.name
        shutil.move(str(item), str(target_path))
        moved_files.append({"file_name": item.name, "from": str(item), "to": str(target_path)})

    return moved_files


def export_to_excel(df: pd.DataFrame, output_dir: str, sheet_name: str) -> str:
    safe_mkdir(output_dir)

    for col in df.columns:
        if pd.api.types.is_datetime64tz_dtype(df[col]):
            df[col] = df[col].dt.tz_convert(None)

    file_name = build_output_file_name()
    out_path  = os.path.join(output_dir, file_name)

    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        df.to_excel(writer, index=False, sheet_name=sheet_name)

    return out_path


def open_file_in_default_app(file_path: str) -> None:
    if not os.path.exists(file_path):
        raise FileNotFoundError(f"File does not exist: {file_path}")
    os.startfile(file_path)

    possible_paths = [
        r"[PATH REDACTED]",

        r"[PATH REDACTED]",
        r"[PATH REDACTED]",
    ]
    for outlook_path in possible_paths:
        if os.path.exists(outlook_path):
            os.startfile(outlook_path)
            return
    try:
        subprocess.Popen(["cmd", "/c", "start", "", "outlook"], shell=False)
    except Exception as e:
        raise RuntimeError(
            "Unable to open Outlook Classic automatically. "
            "Please confirm Outlook Classic is installed on this device."
        ) from e

Excel workbook — VBA & Power Query

The macro-enabled workbook (Weekly Lead Recap Report.xlsm) complements the Python automation layer with two additional components built inside Excel.

Power Query

  • →Connects to the weekly Excel export file as the data source
  • →Separate queries per partner: one per client
  • →Each query filters and shapes data for partner-specific pivot tables
  • →Refreshes automatically as soon as the Excel file is opened — no manual trigger needed

VBA Macros

  • →Partner-specific Save As buttons — exports each partner's pivot summary as a dated file
  • →Partner-specific Send Email buttons — auto-drafts Outlook emails with the recap table and attachment
  • →Dynamic subject lines and email bodies using the weekly date range
  • →4 partners supported: [Partner], [Partner], [Partner], [Partner]
